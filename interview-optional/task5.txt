a) SELECT COUNT(*) AS total_transactions
FROM TRANSACTION t
JOIN ACCOUNT a ON t.ACCOUNT_ID = a.ID
WHERE a.USER_ID = '356789'
  AND t.CATEGORY_ID = '15'
  AND t.POSTED_DATE >= '2013-01-01'
  AND t.POSTED_DATE < '2014-01-01';

b) SELECT t.*
FROM TRANSACTION t
JOIN ACCOUNT a ON t.ACCOUNT_ID = a.ID
WHERE a.USER_ID   = :userId
  AND a.ID         = :accountId
  AND t.POSTED_DATE >= DATE '2026-08-01'
  AND t.POSTED_DATE <  DATE '2026-09-01';

c) SELECT DISTINCT c.ID, c.NAME
FROM CATEGORY c
JOIN TRANSACTION t ON t.CATEGORY_ID = c.ID
JOIN ACCOUNT    a ON t.ACCOUNT_ID   = a.ID
WHERE a.USER_ID = :userId
  AND t.AMOUNT > 0
  AND EXTRACT(YEAR  FROM t.POSTED_DATE) = 2026
  AND EXTRACT(MONTH FROM t.POSTED_DATE) = 8;
